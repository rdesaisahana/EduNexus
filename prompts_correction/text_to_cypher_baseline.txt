###  BASELINE GRAPH SCHEMA ###

graph_schema = """
Nodes:
Student(name)
Teacher(name)
Class(name)
Subject(name)
Department(name)

Relationships:
Student -[:ENROLLED_IN]-> Class
Teacher -[:TEACHES]-> Class
Class -[:FOR_SUBJECT]-> Subject
Teacher -[:BELONGS_TO]-> Department
Teacher -[:HEADS]-> Department
"""
